"""Runs the copy's live gate (tests/browser/live_e2e.py) against the in-memory mutant server of mutrun.py.

The gate writes its evidence inside the throwaway copy (docs/cr-gate/<id>-<engine>-<time>/), never in the repository.
Usage: python gaterun.py <mutants.json> <engine> <id> [<id> ...]
"""
import http.server
import json
import os
import pathlib
import re
import sys
import threading
import time

from mutrun import COPY, handler_for, mutated_text, port_accepts, run_process


def run_gate(mutant, engine):
    body = mutated_text(mutant) if mutant.get("file") else None
    server = http.server.ThreadingHTTPServer(("127.0.0.1", 0), handler_for(mutant if body is not None else None, body))
    port = server.server_address[1]
    threading.Thread(target=server.serve_forever, daemon=True).start()
    out = f"docs/cr-gate/{mutant['id']}-{engine}-{time.strftime('%Y%m%d-%H%M%S')}"
    command = [sys.executable, "-B", "tests/browser/live_e2e.py", "--engine", engine, "--base-url", f"http://127.0.0.1:{port}/", "--out", out]
    try:
        process, output, timed_out, seconds = run_process(command, dict(os.environ, PYTHONDONTWRITEBYTECODE="1"))
    finally:
        server.shutdown()
        server.server_close()
    steps = re.findall(r"^(PASS|FAIL) (\d\d) (.+)$", output, re.M)
    problems = [line.strip() for line in output.splitlines() if line.startswith("    ")][:6]
    verdict_line = next((line for line in output.splitlines() if line.startswith(("GATE", "CONTROL"))), "")
    result = {
        "id": mutant["id"], "kind": "gate", "engine": engine, "pid": process.pid, "exit": process.returncode,
        "timed_out": timed_out, "seconds": seconds, "steps_run": len(steps),
        "first_fail": next((f"{n} {name}" for status, n, name in steps if status == "FAIL"), None),
        "problems": problems, "gate": verdict_line.split(" ->")[0], "out_in_copy": out,
        "server_port_refuses_after": not port_accepts(port),
        "verdict": ("KILLED" if process.returncode != 0 else "SURVIVED") if mutant.get("file") else
                   ("CONTROL GREEN" if process.returncode == 0 else "CONTROL NOT GREEN"),
    }
    print(json.dumps(result), flush=True)


def main():
    mutants = json.loads(pathlib.Path(sys.argv[1]).read_text(encoding="utf-8"))
    engine = sys.argv[2]
    wanted = sys.argv[3:]
    print(json.dumps({"runner_pid": os.getpid(), "engine": engine}), flush=True)
    for mutant in mutants:
        if "all" in wanted or mutant["id"] in wanted:
            run_gate(mutant, engine)


if __name__ == "__main__":
    main()
